function initParticles() {
    const canvas = document.getElementById('particles');
    const ctx = canvas.getContext('2d');
    let particles = [];
    const COUNT = 40;

    function resize() {
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
    }
    resize();
    window.addEventListener('resize', resize);

    class P {
        constructor() { this.reset(); }
        reset() {
            this.x = Math.random() * canvas.width;
            this.y = Math.random() * canvas.height;
            this.sz = Math.random() * 1.5 + 0.3;
            this.vx = (Math.random() - 0.5) * 0.25;
            this.vy = (Math.random() - 0.5) * 0.25;
            this.o = Math.random() * 0.2 + 0.05;
            this.baseO = this.o;
            this.phase = Math.random() * Math.PI * 2;
            this.speed = Math.random() * 0.005 + 0.002;
        }
        update() {
            this.x += this.vx;
            this.y += this.vy;
            this.phase += this.speed;
            this.o = this.baseO * (0.5 + 0.5 * Math.sin(this.phase));
            if (this.x < -10) this.x = canvas.width + 10;
            if (this.x > canvas.width + 10) this.x = -10;
            if (this.y < -10) this.y = canvas.height + 10;
            if (this.y > canvas.height + 10) this.y = -10;
        }
        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.sz, 0, Math.PI * 2);
            ctx.fillStyle = `rgba(0, 221, 179, ${this.o})`;
            ctx.fill();
        }
    }

    for (let i = 0; i < COUNT; i++) particles.push(new P());

    function loop() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        particles.forEach(p => { p.update(); p.draw(); });
        for (let i = 0; i < particles.length; i++) {
            for (let j = i + 1; j < particles.length; j++) {
                const dx = particles[i].x - particles[j].x;
                const dy = particles[i].y - particles[j].y;
                const d = Math.sqrt(dx * dx + dy * dy);
                if (d < 150) {
                    ctx.beginPath();
                    ctx.moveTo(particles[i].x, particles[i].y);
                    ctx.lineTo(particles[j].x, particles[j].y);
                    ctx.strokeStyle = `rgba(0, 221, 179, ${0.03 * (1 - d / 150)})`;
                    ctx.lineWidth = 0.5;
                    ctx.stroke();
                }
            }
        }
        requestAnimationFrame(loop);
    }
    loop();
}

function memeSwap() {
    const a = document.getElementById('meme-a');
    const b = document.getElementById('meme-b');
    if (!a || !b) return;

    const current = a.classList.contains('active') ? a : b;
    const next = current === a ? b : a;

    // Current exits (scale up + fade)
    current.classList.remove('active');
    current.classList.add('exit');

    // Next enters
    memeIdx = (memeIdx + 1) % memeImgs.length;
    next.src = memeImgs[memeIdx];
    next.classList.remove('exit');
    // Force reflow then activate
    void next.offsetWidth;
    next.classList.add('active');

    // Clean up exit class
    setTimeout(() => current.classList.remove('exit'), 300);
}

function buildQuestionList() {
    // 30道常规题，Fisher-Yates洗牌
    const regular = [...APP_DATA.questions];
    for (let i = regular.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [regular[i], regular[j]] = [regular[j], regular[i]];
    }

    // 门控题随机插入到第14-17题位置（约中间）
    const gatePos = 14 + Math.floor(Math.random() * 4);
    const gate = { type: 'gate', ...APP_DATA.gateQuestion };

    // 构建题目列表：前gatePos道常规题 + 门控题 + 剩余常规题
    currentQuestions = [
        ...regular.slice(0, gatePos),
        gate,
        ...regular.slice(gatePos)
    ];
    // 触发题会在门控题选中时动态插入
}

}

            btn.classList.add('selected');
        }
    });

    // 维度加分
    dimScores[q.dim] = (dimScores[q.dim] || 0) + value;

    nextQuestion();
}

    }

// ==========================================
// 结果计算（曼哈顿距离匹配）
// ==========================================
function encodeVector() {
    // 维度分 → H/M/L → 数值向量
    return DIM_ORDER.map(d => {
        const s = dimScores[d] || 0;
        if (s <= 3) return 0; // L
        if (s >= 5) return 2; // H
        return 1; // M
    });
}

function parseVectorStr(str) {
    // "HHH-MMM-..." → [2,2,2,1,1,1,...]
    return str.split('-').join('').split('').map(c => VECTOR_VALUE[c]);
}

function vectorToCode(vec) {
    const labels = vec.map(v => v === 0 ? 'L' : v === 1 ? 'M' : 'H');
    return labels.join('').match(/.{3}/g).join('-');
}

function calculateResult() {
    // 1. 特殊触发优先
    if (isTriggerGuide) {
        const gljl = APP_DATA.specialTypes.find(t => t.code === 'GLJL');
        return {
            ...gljl,
            similarity: 100,
            isSpecial: true,
            userVector: vectorToCode(encodeVector()),
            matchList: []
        };
    }

    // 2. 计算用户向量
    const userVec = encodeVector();

    // 3. 曼哈顿距离匹配
    const matchList = APP_DATA.types.map(type => {
        const typeVec = parseVectorStr(type.vector);
        const distance = userVec.reduce((sum, v, i) => sum + Math.abs(v - typeVec[i]), 0);
        const similarity = (1 - distance / 30) * 100;
        return { ...type, distance, similarity: Number(similarity.toFixed(1)) };
    });

    // 4. 排序：距离最小 → 相似度最高
    matchList.sort((a, b) => a.distance - b.distance || b.similarity - a.similarity);

    // 5. 兜底
    const best = matchList[0];
    if (best.similarity < 60) {
        const mylz = APP_DATA.specialTypes.find(t => t.code === 'MYLZ');
        return {
            ...mylz,
            similarity: best.similarity,
            isSpecial: true,
            userVector: vectorToCode(userVec),
            matchList
        };
    }

    return {
        ...best,
        userVector: vectorToCode(userVec),
        matchList
    };
}

function renderDimensions(userVec) {
    const container = document.getElementById('dim-display');
    if (!container) return;

    let html = '';
    let idx = 0;

    APP_DATA.models.forEach(model => {
        html += `<div class="dim-group">`;
        html += `<div class="dim-group-title">${model.name}</div>`;

        model.dimensions.forEach(dim => {
            const val = userVec[idx];
            const dimName = dim.replace(/^[A-Z]\d\s/, '');
            const level = val === 2 ? '高' : val === 1 ? '中' : '低';
            const cls = val === 2 ? 'dim-h' : val === 1 ? 'dim-m' : 'dim-l';
            const pct = val === 2 ? 100 : val === 1 ? 60 : 25;

            html += `<div class="dim-row">
                <span class="dim-name">${dimName}</span>
                <div class="dim-bar-track">
                    <div class="dim-bar-fill ${cls}" style="width:${pct}%"></div>
                </div>
                <span class="dim-level ${cls}">${level}</span>
            </div>`;
            idx++;
        });

        html += '</div>';
    });

    container.innerHTML = html;
}

function fallbackCopy(text) {
    const ta = document.createElement('textarea');
    ta.value = text; ta.style.cssText = 'position:fixed;opacity:0';

    try { document.execCommand('copy'); showToast('已复制到剪贴板！'); }
    catch { showToast('请手动复制分享'); }
    document.body.removeChild(ta);
}
